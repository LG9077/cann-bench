# Discover operators without hard-coding their names. Empty filter builds all.
set(CANN_BENCH_OPERATORS "$ENV{CANN_BENCH_OPERATORS}" CACHE STRING "Semicolon-separated operator directories")
file(GLOB OP_DIRECTORIES LIST_DIRECTORIES true "${CMAKE_CURRENT_SOURCE_DIR}/*")

# Recreate derived hard links at each configure, including after atomic edits.
file(REMOVE_RECURSE "${CMAKE_CURRENT_SOURCE_DIR}/op_kernel")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/op_kernel")
set(SELECTED_OPERATORS "")
foreach(OP_DIRECTORY IN LISTS OP_DIRECTORIES)
    get_filename_component(OP_NAME "${OP_DIRECTORY}" NAME)
    if(NOT IS_DIRECTORY "${OP_DIRECTORY}" OR OP_NAME STREQUAL "op_kernel" OR OP_NAME MATCHES "^_")
        continue()
    endif()
    if(CANN_BENCH_OPERATORS AND NOT OP_NAME IN_LIST CANN_BENCH_OPERATORS)
        continue()
    endif()
    add_subdirectory("${OP_DIRECTORY}")
    list(APPEND SELECTED_OPERATORS "${OP_NAME}")
endforeach()
foreach(OP_NAME IN LISTS CANN_BENCH_OPERATORS)
    if(NOT OP_NAME IN_LIST SELECTED_OPERATORS)
        message(FATAL_ERROR "Unknown operator directory: ${OP_NAME}")
    endif()
endforeach()
if(NOT SELECTED_OPERATORS)
    message(FATAL_ERROR "No operators selected")
endif()
